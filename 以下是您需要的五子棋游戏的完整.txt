<script>
    (function(){
        // ---------- 游戏配置 ----------
        const BOARD_SIZE = 15;          // 15x15 标准五子棋盘
        const CELL_SIZE = 600 / BOARD_SIZE; // 40px
        const RADIUS = CELL_SIZE * 0.38;    // 棋子半径
        const SNAP_TOLERANCE = 18;          // 点击捕捉容差(px)
        
        let canvas = document.getElementById('gomokuCanvas');
        let ctx = canvas.getContext('2d');

        // 游戏状态
        let board = Array(BOARD_SIZE).fill().map(() => Array(BOARD_SIZE).fill(0)); // 0空 1黑 2白
        let currentPlayer = 1;            // 1: 黑棋(先手)  2: 白棋
        let gameOver = false;
        let winner = null;                 // 'black', 'white', 或 null
        let winCombo = [];                 // 存储胜利连珠坐标 [[row,col], ...]
        let totalMoves = 0;
        
        // 模式: 'twoPlayer' 或 'vsAI'
        let gameMode = 'twoPlayer';        // 默认双人对战
        
        // AI 相关标志
        let isAIPlaying = false;           // 防止AI递归/多重调用
        let pendingAITimer = null;          // AI 延时计时器
        
        // DOM 元素
        let modeSelect = document.getElementById('modeSelect');
        let resetBtn = document.getElementById('resetBtn');
        let turnText = document.getElementById('turnText');
        let turnPieceSample = document.getElementById('turnPieceSample');
        let statusMsgDiv = document.getElementById('statusMsg');
        
        // ---------- 辅助函数 ----------
        function updateUITurnAndStatus() {
            // 更新回合指示器样式
            if (!gameOver) {
                if (currentPlayer === 1) {
                    turnText.innerText = '黑棋落子';
                    turnPieceSample.className = 'piece-sample black';
                } else {
                    turnText.innerText = '白棋落子';
                    turnPieceSample.className = 'piece-sample white';
                }
            } else {
                if (winner === 'black') {
                    turnText.innerText = '胜利 ✨';
                    turnPieceSample.className = 'piece-sample black';
                } else if (winner === 'white') {
                    turnText.innerText = '胜利 ✨';
                    turnPieceSample.className = 'piece-sample white';
                } else {
                    turnText.innerText = '终局';
                }
            }
            
            // 更新状态栏消息
            if (gameOver) {
                if (winner === 'black') statusMsgDiv.innerHTML = '🏆 黑方胜利！ 点击【新一局】继续对战 🏆';
                else if (winner === 'white') statusMsgDiv.innerHTML = '🏆 白方胜利！ 点击【新一局】继续对战 🏆';
                else statusMsgDiv.innerHTML = '🤝 平局！ 棋盘已满 🤝';
            } else {
                if (currentPlayer === 1) statusMsgDiv.innerHTML = '⚫ 黑方回合 · 点击棋盘落子';
                else statusMsgDiv.innerHTML = '⚪ 白方回合 · 点击棋盘落子';
                if (gameMode === 'vsAI' && currentPlayer === 2 && !gameOver) {
                    statusMsgDiv.innerHTML = '🤖 AI 思考中 ...';
                }
            }
        }
        
        // 绘制棋盘 (包括所有棋子 + 胜利高亮)
        function drawBoard() {
            // 背景网格
            ctx.clearRect(0, 0, 600, 600);
            ctx.strokeStyle = '#6b4c3b';
            ctx.lineWidth = 1.8;
            // 绘制网格线
            for (let i = 0; i < BOARD_SIZE; i++) {
                ctx.beginPath();
                ctx.moveTo(i * CELL_SIZE, 0);
                ctx.lineTo(i * CELL_SIZE, 600);
                ctx.stroke();
                ctx.beginPath();
                ctx.moveTo(0, i * CELL_SIZE);
                ctx.lineTo(600, i * CELL_SIZE);
                ctx.stroke();
            }
            // 画星位 (天元/小目 五子棋常见星位 3-3, 11-11, 7-7等)
            let starPoints = [
                [3,3], [11,3], [7,7], [3,11], [11,11]
            ];
            if(BOARD_SIZE === 15) {
                starPoints = [[3,3], [11,3], [7,7], [3,11], [11,11]];
            }
            ctx.fillStyle = '#c9783a';
            for(let p of starPoints) {
                let x = p[0] * CELL_SIZE;
                let y = p[1] * CELL_SIZE;
                ctx.beginPath();
                ctx.arc(x, y, CELL_SIZE*0.1, 0, 2*Math.PI);
                ctx.fillStyle = '#c96f2e';
                ctx.fill();
                ctx.fillStyle = '#e8b35e';
                ctx.arc(x, y, CELL_SIZE*0.06, 0, 2*Math.PI);
                ctx.fill();
            }
            
            // 绘制所有棋子
            for (let row = 0; row < BOARD_SIZE; row++) {
                for (let col = 0; col < BOARD_SIZE; col++) {
                    let val = board[row][col];
                    if (val !== 0) {
                        let x = col * CELL_SIZE;
                        let y = row * CELL_SIZE;
                        let gradient = ctx.createRadialGradient(x+CELL_SIZE*0.3, y+CELL_SIZE*0.3, 5, x+CELL_SIZE*0.5, y+CELL_SIZE*0.5, RADIUS);
                        if (val === 1) { // 黑棋
                            gradient.addColorStop(0, '#2b2b2b');
                            gradient.addColorStop(1, '#111111');
                            ctx.fillStyle = gradient;
                        } else { // 白棋
                            gradient.addColorStop(0, '#fefefe');
                            gradient.addColorStop(1, '#c9cdd2');
                            ctx.fillStyle = gradient;
                        }
                        ctx.beginPath();
                        ctx.arc(x, y, RADIUS, 0, 2 * Math.PI);
                        ctx.fill();
                        ctx.shadowBlur = 0;
                        ctx.strokeStyle = '#5f4c3b';
                        ctx.lineWidth = 0.8;
                        ctx.stroke();
                    }
                }
            }
            
            // 绘制胜利高亮 (金色外圈)
            if (winCombo && winCombo.length > 0) {
                for (let pos of winCombo) {
                    let [row, col] = pos;
                    let x = col * CELL_SIZE;
                    let y = row * CELL_SIZE;
                    ctx.beginPath();
                    ctx.arc(x, y, RADIUS + 5, 0, 2 * Math.PI);
                    ctx.strokeStyle = '#ffcc44';
                    ctx.lineWidth = 4;
                    ctx.shadowBlur = 6;
                    ctx.shadowColor = 'gold';
                    ctx.stroke();
                    ctx.beginPath();
                    ctx.arc(x, y, RADIUS + 2, 0, 2 * Math.PI);
                    ctx.strokeStyle = '#ffaa22';
                    ctx.lineWidth = 2.5;
                    ctx.stroke();
                }
                ctx.shadowBlur = 0;
            }
        }
        
        // 检查胜利 (基于最后一次落子，返回胜利棋子组合)
        function checkWin(row, col, playerVal) {
            if (board[row][col] !== playerVal) return [];
            
            // 四个方向向量: 水平, 垂直, 主对角线(\), 次对角线(/)
            const dirs = [[1,0],[0,1],[1,1],[1,-1]];
            for (let [dx, dy] of dirs) {
                let count = 1;
                let linePositions = [[row, col]];
                // 正方向延伸
                for (let step = 1; step <= 5; step++) {
                    let nr = row + dx * step, nc = col + dy * step;
                    if (nr < 0 || nr >= BOARD_SIZE || nc < 0 || nc >= BOARD_SIZE) break;
                    if (board[nr][nc] === playerVal) {
                        count++;
                        linePositions.push([nr, nc]);
                    } else break;
                }
                // 反方向延伸
                for (let step = 1; step <= 5; step++) {
                    let nr = row - dx * step, nc = col - dy * step;
                    if (nr < 0 || nr >= BOARD_SIZE || nc < 0 || nc >= BOARD_SIZE) break;
                    if (board[nr][nc] === playerVal) {
                        count++;
                        linePositions.push([nr, nc]);
                    } else break;
                }
                if (count >= 5) {
                    // 返回连五的精确组合(取前5个可展示，但全部高亮更有视觉感)
                    return linePositions.slice(0, count);
                }
            }
            return [];
        }
        
        // 检查是否平局 (棋盘满了)
        function isDraw() {
            return totalMoves === BOARD_SIZE * BOARD_SIZE;
        }
        
        // 执行落子核心逻辑 (返回是否成功)
        function placePiece(row, col, playerVal) {
            // 边界合法性、无子、游戏未结束、且下棋方匹配当前player
            if (row < 0 || row >= BOARD_SIZE || col < 0 || col >= BOARD_SIZE) return false;
            if (board[row][col] !== 0) return false;
            if (gameOver) return false;
            if (playerVal !== currentPlayer) return false;
            
            // 落子
            board[row][col] = playerVal;
            totalMoves++;
            
            // 胜利判定
            let winLine = checkWin(row, col, playerVal);
            if (winLine.length >= 5) {
                gameOver = true;
                winner = playerVal === 1 ? 'black' : 'white';
                winCombo = winLine;
                drawBoard();
                updateUITurnAndStatus();
                return true;
            }
            
            // 平局判定
            if (isDraw()) {
                gameOver = true;
                winner = null;
                winCombo = [];
                drawBoard();
                updateUITurnAndStatus();
                return true;
            }
            
            // 没有结束: 切换玩家
            currentPlayer = (currentPlayer === 1) ? 2 : 1;
            drawBoard();
            updateUITurnAndStatus();
            return true;
        }
        
        // 重置游戏 (保留模式)
        function resetGame() {
            // 清除AI定时器
            if (pendingAITimer) {
                clearTimeout(pendingAITimer);
                pendingAITimer = null;
            }
            isAIPlaying = false;
            
            // 重置数据
            board = Array(BOARD_SIZE).fill().map(() => Array(BOARD_SIZE).fill(0));
            currentPlayer = 1;      // 黑棋先手
            gameOver = false;
            winner = null;
            winCombo = [];
            totalMoves = 0;
            
            drawBoard();
            updateUITurnAndStatus();
            
            // 如果当前是人机模式 且 游戏未结束且轮到AI (重置后永远黑棋先, AI不会自动下, 不需要额外动作)
            // 但如果之前AI正等待，重置后取消一切AI事件
        }
        
        // ---------- AI 随机决策 (弱人工智能，随机空位) ----------
        function getRandomEmptyPosition() {
            let empty = [];
            for (let i = 0; i < BOARD_SIZE; i++) {
                for (let j = 0; j < BOARD_SIZE; j++) {
                    if (board[i][j] === 0) empty.push([i, j]);
                }
            }
            if (empty.length === 0) return null;
            let rand = Math.floor(Math.random() * empty.length);
            return empty[rand];
        }
        
        // AI 执行移动 (假设当前轮到AI, 且AI持白棋 playerVal = 2)
        async function aiMove() {
            // 避免并发AI调用 & 防止游戏已结束 & 模式正确 & 当前确实是AI的白方回合 & 游戏未结束
            if (isAIPlaying) return;
            if (gameOver) return;
            if (gameMode !== 'vsAI') return;
            if (currentPlayer !== 2) return;   // AI只在白棋回合行动
            
            isAIPlaying = true;
            
            // 显示思考状态 (更新状态栏)
            updateUITurnAndStatus();
            drawBoard(); // 刷新
            
            // 延时模拟思考感 (200~350ms)
            if (pendingAITimer) clearTimeout(pendingAITimer);
            pendingAITimer = setTimeout(() => {
                pendingAITimer = null;
                // 再次双重校验
                if (gameOver) {
                    isAIPlaying = false;
                    return;
                }
                if (gameMode !== 'vsAI' || currentPlayer !== 2) {
                    isAIPlaying = false;
                    return;
                }
                
                let emptyPos = getRandomEmptyPosition();
                if (!emptyPos) {
                    // 无空位，检查平局
                    if (isDraw() && !gameOver) {
                        gameOver = true;
                        winner = null;
                        winCombo = [];
                        drawBoard();
                        updateUITurnAndStatus();
                    }
                    isAIPlaying = false;
                    return;
                }
                let [row, col] = emptyPos;
                // 调用通用落子 (AI执白 = 2)
                let success = placePiece(row, col, 2);
                
                // 落子后重绘及更新状态
                drawBoard();
                updateUITurnAndStatus();
                
                // 如果游戏未结束 & 未平局 & 模式仍为人机 & 且当前玩家又变成了AI（正常情况下白方落子后切换至黑方，除非胜利或结束）
                // 这里不需要额外调用，因为aiMove仅执行一次, 等待玩家回合玩家落子后会自动触发新的AI回合
                isAIPlaying = false;
                
                // 如果游戏在AI移动后结束, 不需要额外操作
                if (!gameOver && gameMode === 'vsAI' && currentPlayer === 2) {
                    // 极少数情况(比如某次移动后没有切换玩家？正常情况下placePiece会切换玩家，所以不会发生。为了稳健，若出现轮到AI则递归调用)
                    // 添加递归避免状态错误，但绝大情况不会进入，防止死循环用setTimeout
                    if (!isAIPlaying && !gameOver && currentPlayer === 2) {
                        setTimeout(() => { if(!gameOver && gameMode==='vsAI' && currentPlayer===2) aiMove(); }, 50);
                    }
                }
            }, 220);
        }
        
        // 玩家移动处理 (根据当前模式调用)
        function handlePlayerMove(row, col) {
            if (gameOver) {
                return false;
            }
            // 双人模式: 任何人都可以下，但必须符合当前玩家
            if (gameMode === 'twoPlayer') {
                let success = placePiece(row, col, currentPlayer);
                if (success) {
                    drawBoard();
                    updateUITurnAndStatus();
                    return true;
                }
                return false;
            } 
            // 人机模式: 只有黑棋(玩家)可以主动下子；白棋是AI托管，玩家不能直接落白子
            else if (gameMode === 'vsAI') {
                // 人机模式玩家只能控制黑棋(1)，且必须游戏未结束且轮到黑棋
                if (currentPlayer !== 1) {
                    statusMsgDiv.innerHTML = '🤖 等待AI落子，请稍后...';
                    setTimeout(() => updateUITurnAndStatus(), 800);
                    return false;
                }
                let success = placePiece(row, col, 1);
                if (success) {
                    drawBoard();
                    updateUITurnAndStatus();
                    // 玩家落子后，如果游戏未结束 && 模式人机 && 当前轮到白棋(AI) && 无gameover，启动AI
                    if (!gameOver && gameMode === 'vsAI' && currentPlayer === 2) {
                        // 避免多个AI调用
                        if (!isAIPlaying) {
                            aiMove();
                        }
                    }
                    return true;
                }
                return false;
            }
            return false;
        }
        
        // ---------- 画布点击坐标转换 ----------
        function getBoardPositionFromClick(clientX, clientY) {
            const rect = canvas.getBoundingClientRect();
            const scaleX = canvas.width / rect.width;   // 通常canvas实际像素600/显示宽高比，若无缩放即1
            const scaleY = canvas.height / rect.height;
            const canvasX = (clientX - rect.left) * scaleX;
            const canvasY = (clientY - rect.top) * scaleY;
            
            // 计算最近交点 (网格坐标)
            let minDist = SNAP_TOLERANCE;
            let targetRow = -1, targetCol = -1;
            for (let i = 0; i < BOARD_SIZE; i++) {
                for (let j = 0; j < BOARD_SIZE; j++) {
                    let centerX = j * CELL_SIZE;
                    let centerY = i * CELL_SIZE;
                    let dx = canvasX - centerX;
                    let dy = canvasY - centerY;
                    let dist = Math.hypot(dx, dy);
                    if (dist < minDist) {
                        minDist = dist;
                        targetRow = i;
                        targetCol = j;
                    }
                }
            }
            if (targetRow !== -1 && targetCol !== -1 && board[targetRow][targetCol] === 0) {
                return { row: targetRow, col: targetCol };
            }
            return null;
        }
        
        function onCanvasClick(e) {
            if (gameOver) {
                statusMsgDiv.innerHTML = winner ? '🏁 游戏已结束，点击【新一局】继续挑战' : '🤝 平局结束，点击新一局';
                return;
            }
            // AI 正在思考时禁止点击
            if (isAIPlaying) {
                statusMsgDiv.innerHTML = '🤖 AI 走棋中，请稍后...';
                return;
            }
            const pos = getBoardPositionFromClick(e.clientX, e.clientY);
            if (!pos) return;
            
            const { row, col } = pos;
            handlePlayerMove(row, col);
        }
        
        // ---------- 模式切换 ----------
        function switchMode() {
            let newMode = modeSelect.value;
            if (newMode === gameMode) {
                // 相同模式重置只重置棋盘即可
                resetGame();
                return;
            }
            gameMode = newMode;
            // 重置所有状态，清除AI调用
            if (pendingAITimer) {
                clearTimeout(pendingAITimer);
                pendingAITimer = null;
            }
            isAIPlaying = false;
            resetGame();  // 重置棋盘，currentPlayer = 1 黑棋先手
            
            // 额外状态提示
            if (gameMode === 'vsAI') {
                statusMsgDiv.innerHTML = '🤖 人机模式：您执黑棋(先手)，AI执白棋，点击棋盘落子';
            } else {
                statusMsgDiv.innerHTML = '👥 双人对战模式：黑棋先手，轮流落子';
            }
            updateUITurnAndStatus();
            drawBoard();
        }
        
        // ---------- 重置（额外确保AI取消）----------
        function onReset() {
            if (pendingAITimer) {
                clearTimeout(pendingAITimer);
                pendingAITimer = null;
            }
            isAIPlaying = false;
            resetGame();
            if (gameMode === 'vsAI' && currentPlayer === 2) {
                // 重置后永远当前玩家是黑棋1，所以不需要AI主动下，但如果有意外情况不处理
            }
            statusMsgDiv.innerHTML = gameMode === 'vsAI' ? '🔄 新对局：黑棋(玩家)先走' : '🔄 新对局：黑棋先行';
            drawBoard();
        }
        
        // ---------- 事件绑定 ----------
        function bindEvents() {
            canvas.addEventListener('click', onCanvasClick);
            resetBtn.addEventListener('click', onReset);
            modeSelect.addEventListener('change', switchMode);
        }
        
        // 初始化全部
        function init() {
            bindEvents();
            resetGame();
            drawBoard();
            updateUITurnAndStatus();
        }
        
        init();
    })();
</script>
